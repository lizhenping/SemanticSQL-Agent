class TripleOutput(BaseModel):
    """标准三元组输出结构 - 所有工具的统一输出格式"""
    subject: str = Field(description="主体：表名、字段名、SQL语句等")
    predicate: str = Field(description="关系：has_column、belongs_to、generates_sql等")
    object: str = Field(description="客体：具体值、结构化数据、分析结果等")
    confidence: float = Field(default=1.0, description="置信度：0.0-1.0")
    source: str = Field(default="", description="数据来源：工具名称")
    timestamp: datetime = Field(default_factory=datetime.now, description="生成时间")
```

class AgentState(TypedDict):
    """极简智能体状态 - ReAct循环的核心状态"""

    current_input: str                   # 当前用户输入
    conversation_id: str                 # 会话ID
    user_id: Optional[str]              # 用户ID
    session_metadata: Dict[str, Any]     # 会话元数据

class ExecutionContext(BaseModel):
    """执行上下文 - 工具执行环境信息"""
    database_config: Optional[Dict[str, Any]] = None
    user_preferences: Dict[str, Any] = Field(default_factory=dict)
    execution_limits: Dict[str, Any] = Field(default_factory=dict)
    debug_mode: bool = Field(default=False)
```
