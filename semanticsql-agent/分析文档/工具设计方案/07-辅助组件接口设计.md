### 2.2 模板管理器 (prompts/manager.py)
```python
from jinja2 import Template
from typing import Dict, Any, List
from core.memory import format_memory_text

class TemplateManager:
    """极简模板管理器"""
    
    def __init__(self):
        self.templates = {}
        self._load_default_templates()
    
    def _load_default_templates(self):
        """加载默认模板"""
        from .system_template import DEFAULT_SYSTEM_TEMPLATE, SQL_AGENT_TEMPLATE
        self.templates["default"] = Template(DEFAULT_SYSTEM_TEMPLATE)
        self.templates["sql_agent"] = Template(SQL_AGENT_TEMPLATE)
    
    def render(self, template_name: str, **kwargs) -> str:
        """渲染模板"""
        template = self.templates.get(template_name)
        if not template:
            raise ValueError(f"模板不存在: {template_name}")
        
        return template.render(**kwargs)
    
    def render_system_message(self, 
                             template_name: str,
                             memory: List,
                             tools: List, 
                             **kwargs) -> str:
        """渲染系统消息"""
        return self.render(
            template_name,
            memory_text=format_memory_text(memory),
            memory_count=len(memory),
            tools=[{"name": tool.name, "description": tool.description} for tool in tools],
            **kwargs
        )

# 全局模板管理器
template_manager = TemplateManager()
```

## 3. 工具模块 (utils/)

### 3.1 数据库管理 (utils/database.py)
```python
import pymysql
from typing import Dict, Any, List, Optional
from config.database import DatabaseConfig

class DatabaseManager:
    """极简数据库管理器"""
    
    def __init__(self, config: DatabaseConfig):
        self.config = config
        self._connection = None
    
    def connect(self):
        """创建数据库连接"""
        self._connection = pymysql.connect(
            host=self.config.host,
            port=self.config.port,
            user=self.config.username,
            password=self.config.password,
            database=self.config.database,
            charset=self.config.charset
        )
        return self._connection
    
    def execute_query(self, sql: str, limit: int = 100) -> List[Dict[str, Any]]:
        """执行查询"""
        if not self._connection:
            self.connect()
        
        with self._connection.cursor(pymysql.cursors.DictCursor) as cursor:
            cursor.execute(sql)
            return cursor.fetchmany(limit)
    
    def get_schema_info(self) -> Dict[str, List[Dict[str, Any]]]:
        """获取数据库schema信息"""
        schema_query = """
        SELECT TABLE_NAME, COLUMN_NAME, DATA_TYPE, IS_NULLABLE, COLUMN_KEY
        FROM INFORMATION_SCHEMA.COLUMNS 
        WHERE TABLE_SCHEMA = %s
        ORDER BY TABLE_NAME, ORDINAL_POSITION
        """
        
        if not self._connection:
            self.connect()
            
        with self._connection.cursor(pymysql.cursors.DictCursor) as cursor:
            cursor.execute(schema_query, (self.config.database,))
            results = cursor.fetchall()
        
        # 按表名组织结果
        schema = {}
        for row in results:
            table_name = row['TABLE_NAME']
            if table_name not in schema:
                schema[table_name] = []
            schema[table_name].append(row)
        
        return schema
    
    def close(self):
        """关闭连接"""
        if self._connection:
            self._connection.close()
            self._connection = None
```

### 3.2 LLM客户端 (utils/llm_client.py)
```python
from typing import Any, List, Dict, Optional
from langchain_openai import ChatOpenAI

def create_llm_client(config: Dict[str, Any]) -> Any:
    """创建LLM客户端 - 极简工厂函数"""
    
    model = config.get("model", "gpt-4")
    api_key = config["api_key"]
    base_url = config.get("base_url")
    temperature = config.get("temperature", 0.7)
    
    llm = ChatOpenAI(
        model=model,
        api_key=api_key,
        base_url=base_url,
        temperature=temperature
    )
    
    return llm

class LLMClient:
    """LLM客户端封装类（如果需要更多控制）"""
    
    def __init__(self, config: Dict[str, Any]):
        self.llm = create_llm_client(config)
    
    def invoke(self, messages: List) -> Any:
        """调用LLM"""
        return self.llm.invoke(messages)
    
    def bind_tools(self, tools: List) -> Any:
        """绑定工具"""
        return self.llm.bind_tools(tools)
```

## 4. CLI接口 (cli.py)

### 4.1 命令行接口
```python
import click
from typing import Dict, Any
from config.settings import settings
from config.database import DatabaseConfig
from agent.factory import create_semantic_sql_agent
from tools import get_all_tools

@click.group()
def cli():
    """SemanticSQL Agent 命令行工具"""
    pass

@cli.command()
@click.option('--host', required=True, help='数据库主机')
@click.option('--database', required=True, help='数据库名')
@click.option('--username', required=True, help='用户名')
@click.option('--password', required=True, help='密码')
@click.option('--port', default=3306, help='端口号')
@click.option('--question', help='自然语言问题')
def analyze(host, database, username, password, port, question):
    """分析数据库并生成SQL"""
    
    # 创建数据库配置
    db_config = {
        "host": host,
        "database": database, 
        "username": username,
        "password": password,
        "port": port
    }
    
    # 创建智能体
    agent = create_semantic_sql_agent(
        llm_config={
            "model": settings.llm_model,
            "api_key": settings.llm_api_key,
            "base_url": settings.llm_base_url,
            "temperature": settings.llm_temperature
        },
        tools=get_all_tools()
    )
    
    # 执行分析
    result = agent.analyze_and_generate_sql(
        database_params=db_config,
        user_question=question
    )
    
    # 显示结果
    click.echo(f"✅ 分析完成，生成了 {len(result['memory'])} 个知识三元组")
    
    # 显示最终的SQL（如果有）
    for s, p, o in result["memory"]:
        if p == "对应SQL":
            click.echo(f"🔍 生成的SQL: {o}")
            break

@cli.command() 
@click.option('--config', help='配置文件路径')
def interactive(config):
    """交互式模式"""
    click.echo("🚀 启动 SemanticSQL Agent 交互模式")
    click.echo("输入 'help' 查看帮助，输入 'quit' 退出")
    
    # 这里可以实现交互式对话逻辑
    while True:
        user_input = click.prompt("SQL Agent", type=str)
        if user_input.lower() in ['quit', 'exit']:
            break
        
        # 处理用户输入...
        click.echo(f"处理中: {user_input}")

if __name__ == "__main__":
    cli()
```

**极简设计特点**：
1. **配置简化**：只保留必要的配置项
2. **模板管理**：基于Jinja2的简单模板系统
3. **数据库工具**：基础的连接和查询功能
4. **LLM集成**：简单的工厂函数创建客户端
5. **CLI接口**：基于Click的命令行工具
6. **工具集成**：简单的工具导入和组织
