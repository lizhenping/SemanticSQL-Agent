# 工具基类极简接口设计（集成图记忆）

## 1. 记忆数据模型 (tools/schemas.py)

### 1.1 语义三元组数据模型
```python
from pydantic import BaseModel, Field
from typing import Optional, Tuple
from langchain_neo4j import Neo4jGraph
from langchain_community.graphs.graph_document import GraphDocument, Node, Relationship

class SemanticTriple(BaseModel):
    """语义三元组模型 - 工具记忆的核心数据结构"""
    subject: str = Field(description="主体实体")
    predicate: str = Field(description="关系谓词")
    object: str = Field(description="客体实体")
    
    # 扩展属性
    subject_type: str = Field(default="Entity", description="主体类型")
    object_type: str = Field(default="Entity", description="客体类型") 
    confidence: Optional[float] = Field(default=None, description="置信度（可选）")
    source_tool: str = Field(default="", description="来源工具")
    
    def to_simple_tuple(self) -> Tuple[str, str, str]:
        """转换为简单三元组，保持向后兼容"""
        return (self.subject, self.predicate, self.object)
```

## 2. 工具基类接口 (tools/base_tool.py)

### 2.1 记忆管理功能集成工具基类
```python
from langchain.tools import BaseTool
from typing import Optional, List, Union
from typing_extensions import TypedDict
from .schemas import SemanticTriple

class AgentState(TypedDict):
    """智能体状态 - 极简设计"""
    memory: List[SemanticTriple]      # 语义三元组记忆
    current_input: str                # 当前用户输入

class BaseSemanticSQLTool(BaseTool):
    """SemanticSQL工具基类 - 集成完整记忆管理功能"""
    
    def __init__(self, neo4j_graph: Optional[Neo4jGraph] = None, **kwargs):
        super().__init__(**kwargs)
        self.neo4j_graph = neo4j_graph  # 可选的Neo4j图实例
    
    def execute_with_state(self, state: AgentState, **kwargs) -> str:
        """核心执行方法 - 子类必须实现"""
        raise NotImplementedError
    
    def _run(self, state: AgentState, **kwargs) -> str:
        """统一执行入口"""
        return self.execute_with_state(state, **kwargs)
    
    # 记忆管理核心方法
    def add_triple(self, 
                   memory: List[SemanticTriple],
                   subject: str,
                   predicate: str, 
                   object: str,
                   subject_type: str = "Entity",
                   object_type: str = "Entity",
                   confidence: Optional[float] = None,
                   source_tool: str = "") -> None:
        """添加语义三元组到记忆 + 可选Neo4j持久化"""
        triple = SemanticTriple(
            subject=subject,
            predicate=predicate,
            object=object,
            subject_type=subject_type,
            object_type=object_type,
            confidence=confidence,
            source_tool=source_tool or self.name
        )
        memory.append(triple)
        
        # 可选：同步到Neo4j
        if self.neo4j_graph:
            self.persist_triples_to_neo4j([triple])
    
    def add_knowledge_triple(self, 
                           state: AgentState,
                           subject: str, 
                           predicate: str, 
                           object: str,
                           subject_type: str = "Entity",
                           object_type: str = "Entity",
                           confidence: Optional[float] = None) -> None:
        """便捷方法：添加知识三元组到状态记忆"""
        self.add_triple(
            memory=state["memory"],
            subject=subject,
            predicate=predicate,
            object=object,
            subject_type=subject_type,
            object_type=object_type,
            confidence=confidence,
            source_tool=self.name
        )
    
    def query_by_subject(self, memory: List[SemanticTriple], subject: str) -> List[SemanticTriple]:
        """按主体查询三元组"""
        return [triple for triple in memory if triple.subject == subject]
    
    def query_by_source(self, memory: List[SemanticTriple], source_tool: str) -> List[SemanticTriple]:
        """按来源工具查询三元组"""
        return [triple for triple in memory if triple.source_tool == source_tool]
    
    def format_memory_text(self, memory: List[SemanticTriple]) -> str:
        """格式化记忆为LLM可读文本"""
        if not memory:
            return "记忆为空，需要开始分析"
        
        lines = []
        for triple in memory:
            confidence_str = f" (置信度:{triple.confidence:.2f})" if triple.confidence else ""
            tool_str = f" [{triple.source_tool}]" if triple.source_tool else ""
            lines.append(f"- {triple.subject} {triple.predicate} {triple.object}{confidence_str}{tool_str}")
        
        return "\n".join(lines)
    
    # Neo4j集成方法
    def convert_to_graph_document(self, triples: List[SemanticTriple], 
                                 source: str = "SemanticSQL-Agent") -> GraphDocument:
        """将语义三元组转换为LangChain GraphDocument"""
        nodes = []
        relationships = []
        node_set = set()
        
        for triple in triples:
            # 创建主体节点
            if triple.subject not in node_set:
                subject_node = Node(
                    id=triple.subject,
                    type=triple.subject_type,
                    properties={"name": triple.subject, "source": triple.source_tool}
                )
                nodes.append(subject_node)
                node_set.add(triple.subject)
            
            # 创建客体节点
            if triple.object not in node_set:
                object_node = Node(
                    id=triple.object,
                    type=triple.object_type,
                    properties={"name": triple.object, "source": triple.source_tool}
                )
                nodes.append(object_node)
                node_set.add(triple.object)
            
            # 创建关系
            relationship = Relationship(
                source=Node(id=triple.subject, type=triple.subject_type),
                target=Node(id=triple.object, type=triple.object_type),
                type=triple.predicate.upper().replace(" ", "_"),
                properties={
                    "confidence": triple.confidence,
                    "source_tool": triple.source_tool
                }
            )
            relationships.append(relationship)
        
        return GraphDocument(
            nodes=nodes,
            relationships=relationships,
            source=source
        )
    
    def persist_triples_to_neo4j(self, triples: List[SemanticTriple]) -> None:
        """将语义三元组持久化到Neo4j"""
        if not triples or not self.neo4j_graph:
            return
        
        graph_doc = self.convert_to_graph_document(triples)
        self.neo4j_graph.add_graph_documents([graph_doc])
    
```

## 3. 使用示例

### 3.1 基础使用（仅内存）
```python
from tools.base_tool import BaseSemanticSQLTool, AgentState
from langchain_core.tools import tool

@tool
def schema_extraction(state: AgentState) -> str:
    """数据库结构提取工具 - 使用集成记忆管理"""
    
    # 创建工具实例
    tool = BaseSemanticSQLTool()
    tool.name = "schema_extraction"
    
    # 模拟数据库分析
    tables = {"users": ["id", "name"], "orders": ["id", "user_id"]}
    
    for table_name, columns in tables.items():
        # 添加表信息
        tool.add_knowledge_triple(
            state,
            subject="数据库",
            predicate="包含表",
            object=table_name,
            subject_type="Database",
            object_type="Table"
        )
        
        # 添加字段信息
        for col in columns:
            tool.add_knowledge_triple(
                state,
                subject=table_name,
                predicate="包含字段",
                object=col,
                subject_type="Table",
                object_type="Column",
                confidence=0.95
            )
    
    return tool.create_result(
        summary=f"提取了{len(tables)}个表的结构信息",
        triples=tool.query_by_source(state["memory"], "schema_extraction")
    )
```

### 3.2 Neo4j集成使用

```python
        object_type="Table"

        summary="分析完成并同步到Neo4j",
        triples=state["memory"][-1:]  # 返回最新添加的三元组
    )
```

## 4. 设计优势

**基于就近原则的记忆集成设计特点**：

### 4.1 架构优势
1. **就近原则**：记忆数据模型和管理功能都在tools模块，便于维护
2. **减少依赖**：避免tools模块依赖core模块的复杂数据类型
3. **职责清晰**：BaseSemanticSQLTool成为记忆功能的完整载体
4. **扩展方便**：修改记忆结构只需在tools模块内操作

### 4.2 功能优势  
1. **完整记忆管理**：add_triple(), query_by_subject(), format_memory_text()
2. **可选Neo4j集成**：GraphDocument转换和持久化功能
3. **原型友好**：可选的confidence字段，灵活的Neo4j配置
4. **查询精准**：只支持subject和source查询，符合实际需求

### 4.3 使用优势
1. **继承即获得**：所有具体工具继承BaseSemanticSQLTool自动获得记忆能力
2. **统一接口**：记忆操作、Neo4j集成、结果创建都在基类中
3. **开发高效**：工具实现只需关注业务逻辑，记忆管理自动处理
4. **符合ReAct**：仍然是LLM驱动的工具选择，不影响智能编排
