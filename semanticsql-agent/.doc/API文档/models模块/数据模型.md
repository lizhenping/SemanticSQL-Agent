# 数据模型 API 文档

SemanticSQL Agent 使用的所有数据模型定义，基于 Pydantic 实现。

## 核心模型

### QueryScenario

```python
from pydantic import BaseModel, Field
from typing import List, Optional
from datetime import datetime

class QueryScenario(BaseModel):
    """
    查询场景定义
    
    代表一个特定的业务查询场景，用于生成相关的问题和 SQL。
    """
    id: str = Field(description="场景唯一标识")
    category: str = Field(description="场景类别：基础查询/统计分析/关联查询等")
    business_purpose: str = Field(description="业务目的描述")
    description: str = Field(description="场景详细描述")
    difficulty: str = Field(description="难度级别：easy/medium/hard/expert")
    tables: List[str] = Field(description="涉及的表")
    operations: List[str] = Field(description="包含的 SQL 操作")
    
    class Config:
        schema_extra = {
            "example": {
                "id": "scenario_001",
                "category": "统计分析",
                "business_purpose": "分析销售趋势",
                "description": "统计最近30天每日的订单数量和销售额",
                "difficulty": "medium",
                "tables": ["orders"],
                "operations": ["GROUP BY", "AGGREGATE", "DATE_FUNCTION"]
            }
        }
```

### SQLQueryResult

```python
class SQLQueryResult(BaseModel):
    """
    SQL 查询结果
    
    包含生成的 SQL 及其执行结果。
    """
    id: str = Field(description="结果ID")
    question: str = Field(description="原始自然语言问题")
    sql: str = Field(description="生成的 SQL 查询")
    
    # 执行信息
    execution_success: bool = Field(description="是否执行成功")
    execution_time: float = Field(description="执行时间（秒）")
    row_count: int = Field(default=0, description="返回行数")
    result: Optional[List[Dict]] = Field(default=None, description="查询结果（前N行）")
    
    # 错误信息
    error: Optional[str] = Field(default=None, description="错误信息")
    error_type: Optional[str] = Field(default=None, description="错误类型")
    
    # 元数据
    confidence: float = Field(ge=0, le=1, description="生成置信度")
    complexity: str = Field(description="查询复杂度")
    tables_used: List[str] = Field(description="使用的表")
    timestamp: datetime = Field(default_factory=datetime.now)
    
    def is_valid(self) -> bool:
        """检查结果是否有效"""
        return self.execution_success and self.error is None
```

### TrainingExample

```python
class TrainingExample(BaseModel):
    """
    训练数据示例
    
    NL2SQL 训练数据的标准格式。
    """
    id: str = Field(description="示例ID")
    scenario_id: str = Field(description="关联的场景ID")
    
    # 核心数据
    question: str = Field(description="自然语言问题")
    sql: str = Field(description="对应的 SQL 查询")
    
    # 验证信息
    validated: bool = Field(description="是否已验证")
    validation_result: Optional[Dict] = Field(description="验证结果")
    
    # 质量信息
    quality_score: float = Field(ge=0, le=1, description="质量分数")
    difficulty: str = Field(description="难度级别")
    
    # 元数据
    database_info: Dict[str, Any] = Field(description="数据库信息")
    created_at: datetime = Field(default_factory=datetime.now)
    tags: List[str] = Field(default=[], description="标签")
    
    def to_training_format(self) -> Dict[str, str]:
        """转换为训练格式"""
        return {
            "input": self.question,
            "output": self.sql,
            "metadata": {
                "difficulty": self.difficulty,
                "quality": self.quality_score
            }
        }
```

## 分析结果模型

### SchemaInfo

```python
class TableInfo(BaseModel):
    """表信息"""
    name: str
    comment: Optional[str] = None
    engine: str = "InnoDB"
    row_count: Optional[int] = None
    created_at: Optional[datetime] = None

class ColumnInfo(BaseModel):
    """列信息"""
    table_name: str
    column_name: str
    data_type: str
    is_nullable: bool = True
    default_value: Optional[str] = None
    comment: Optional[str] = None
    is_primary_key: bool = False
    is_foreign_key: bool = False

class SchemaInfo(BaseModel):
    """数据库 Schema 信息"""
    database_name: str
    tables: List[TableInfo]
    columns: List[ColumnInfo]
    foreign_keys: List[Dict[str, str]]
    indexes: List[Dict[str, Any]]
    
    def get_table_columns(self, table_name: str) -> List[ColumnInfo]:
        """获取指定表的所有列"""
        return [c for c in self.columns if c.table_name == table_name]
```

```python
from enum import Enum

class StepType(str, Enum):
    THOUGHT = "thought"
    ACTION = "action"
    OBSERVATION = "observation"
    RESULT = "result"

class AgentStep(BaseModel):
    """Agent 执行步骤"""
    step_id: str
    step_type: StepType
    content: str
    tool_name: Optional[str] = None
    tool_input: Optional[Dict[str, Any]] = None
    tool_output: Optional[Dict[str, Any]] = None
    timestamp: datetime = Field(default_factory=datetime.now)
    duration: Optional[float] = None
    error: Optional[str] = None
```

### AgentExecution

```python
class AgentExecution(BaseModel):
    """Agent 完整执行记录"""
    execution_id: str
    task: str = Field(description="任务描述")
    status: str = Field(description="状态：running/completed/failed")
    
    # 执行步骤
    steps: List[AgentStep] = Field(default=[])
    
    # 时间信息
    started_at: datetime
    completed_at: Optional[datetime] = None
    total_duration: Optional[float] = None
    
    # 结果
    final_result: Optional[Dict[str, Any]] = None
    error: Optional[str] = None
    
    # 统计
    total_steps: int = 0
    tool_calls: int = 0
    llm_calls: int = 0
    
    def add_step(self, step: AgentStep):
        """添加执行步骤"""
        self.steps.append(step)
        self.total_steps += 1
        if step.tool_name:
            self.tool_calls += 1
```

## 统计模型

### GenerationStatistics

```python
class GenerationStatistics(BaseModel):
    """生成统计信息"""
    total_requested: int = Field(description="请求生成数量")
    total_generated: int = Field(description="实际生成数量")
    success_count: int = Field(description="成功数量")
    failed_count: int = Field(description="失败数量")
    
    # 难度分布
    difficulty_distribution: Dict[str, int] = Field(
        default={},
        description="难度分布"
    )
    
    # 表覆盖
    table_coverage: Dict[str, int] = Field(
        default={},
        description="各表使用次数"
    )
    
    # 操作类型分布
    operation_distribution: Dict[str, int] = Field(
        default={},
        description="SQL操作分布"
    )
    
    # 性能指标
    avg_generation_time: float = Field(description="平均生成时间")
    avg_execution_time: float = Field(description="平均执行时间")
    
    # 质量指标
    avg_quality_score: float = Field(description="平均质量分数")
    validation_pass_rate: float = Field(description="验证通过率")
    
    def success_rate(self) -> float:
        """计算成功率"""
        if self.total_generated == 0:
            return 0.0
        return self.success_count / self.total_generated
```

### 批量处理训练数据

```python
# 生成训练数据
examples = []
for i in range(100):
    example = TrainingExample(
        id=f"train_{i:04d}",
        scenario_id="scenario_001",
        question=f"查询问题 {i}",
        sql=f"SELECT * FROM table_{i}",
        validated=True,
        quality_score=0.85,
        difficulty="medium",
        database_info={"name": "mydb", "version": "5.7"}
    )
    examples.append(example)

# 导出为训练格式
training_data = [ex.to_training_format() for ex in examples]

# 保存到文件
import json
with open("training_data.jsonl", "w") as f:
    for item in training_data:
        f.write(json.dumps(item) + "\n")
```

## 模型验证

### 自定义验证

```python
from pydantic import validator

class StrictTrainingExample(TrainingExample):
    """严格的训练示例"""
    
    @validator('sql')
    def validate_sql(cls, v):
        """验证 SQL 语法"""
        if not v.strip().upper().startswith(('SELECT', 'INSERT', 'UPDATE', 'DELETE')):
            raise ValueError('SQL must start with a valid operation')
        return v
    
    @validator('quality_score')
    def validate_quality(cls, v):
        """确保质量分数在合理范围"""
        if v < 0.5:
            raise ValueError('Quality score too low for training')
        return v
```

## 注意事项

1. **版本兼容**：使用 Pydantic V1 语法（LangChain 0.1.x 要求）
2. **类型安全**：所有字段都有明确的类型定义
3. **验证规则**：合理使用验证器确保数据质量
4. **性能考虑**：大量数据时注意内存使用
5. **序列化**：注意时间类型的序列化格式
