#### 2.1 分析工具示例（schema_extraction_tool.py）
```python
from typing import Dict, Any, Type
from langchain.tools import BaseTool
from pydantic import BaseModel, Field
from semanticsql_agent.models.exceptions import (
    ToolExecutionError, 
    DatabaseConnectionError,
    SchemaExtractionError
)

class SchemaExtractionInput(BaseModel):
    """Schema提取输入"""
    database_name: str = Field(description="数据库名称")

class SchemaExtractionTool(BaseTool):
    name = "schema_extraction"
    description = "提取数据库的完整结构信息"
    args_schema: Type[BaseModel] = SchemaExtractionInput
    
    def __init__(self, db_manager):
        super().__init__()
        self.db_manager = db_manager
    
    def _run(self, database_name: str) -> Dict[str, Any]:
        """执行schema提取"""
        try:
            # 原有的提取逻辑
            schema_info = self.db_manager.extract_schema(database_name)
            return schema_info
        except DatabaseConnectionError as e:
            raise ToolExecutionError(
                tool_name=self.name,
                reason=f"数据库连接失败: {e.message}"
            )
        except Exception as e:
            raise SchemaExtractionError(
                database=database_name,
                error=str(e)
            )
```
