## 第四阶段：重构 Agent（12小时）

### 4.1 重写 base_agent.py
```python
from abc import ABC, abstractmethod
from typing import List, Dict, Any, Optional
from langchain.agents import AgentExecutor, create_react_agent
from langchain.prompts import ChatPromptTemplate
from langchain.chat_models import ChatOpenAI
from langchain.tools import BaseTool
from langchain.memory import BaseMemory

class BaseAgent(ABC):
    """基于 LangChain 的基础 Agent"""
    
    def __init__(self, settings: Settings, db_config: DatabaseConfig):
        self.settings = settings
        self.db_config = db_config
        
        # 初始化 LLM
        self.llm = ChatOpenAI(
            model=settings.llm_model,
            temperature=settings.llm_temperature,
            openai_api_base=settings.llm_base_url,
            openai_api_key=settings.llm_api_key
        )
        
        # 初始化工具
        self.tools = self._initialize_tools()
        
        # 初始化记忆
        self.memory = self._initialize_memory()
        
        # 创建 Agent
        self.agent_executor = self._create_agent()
    
    @abstractmethod
    def _initialize_tools(self) -> List[BaseTool]:
        """初始化工具列表"""
        pass
    
    @abstractmethod
    def _initialize_memory(self) -> BaseMemory:
        """初始化记忆系统"""
        pass
    
    def _create_agent(self) -> AgentExecutor:
        """创建 LangChain Agent"""
        # 创建提示词模板
        prompt = self._create_prompt()
        
        # 创建 ReAct agent
        agent = create_react_agent(
            llm=self.llm,
            tools=self.tools,
            prompt=prompt
        )
        
        # 创建执行器
        return AgentExecutor(
            agent=agent,
            tools=self.tools,
            memory=self.memory,
            verbose=True,
            max_iterations=self.settings.max_steps
        )
```

from typing import List, Dict, Any, Optional
from langchain.tools import BaseTool
from utils.memory import DatabaseAnalysisMemory
from utils.database import DatabaseManager

# 导入所有工具
from tools.analysis_tools import (
    SchemaExtractionTool, DomainAnalysisTool,
    FieldClassificationTool, ColumnMeaningTool,
    TableMeaningTool, ERAnalysisTool
)
from tools.generation_tools import (
    ScenarioTool, OperationSelectionTool,
    QuestionGenerationTool, SQLGenerationTool
)
from tools.validation_tools import (
    SQLValidationTool, SQLExecutionTool
)
from tools.reflection_tools import SQLReflectionTool
from tools.thinking_tools import SequentialThinkingTool

class SQLAgent(BaseAgent):
    """SQL 生成智能体 - 支持批量训练数据生成"""
    
    def _initialize_tools(self) -> List[BaseTool]:
        """初始化所有工具"""
        db_manager = DatabaseManager(self.db_config)
        
        tools = [
            # 分析工具
            SchemaExtractionTool(db_manager=db_manager),
            DomainAnalysisTool(),
            FieldClassificationTool(),
            ColumnMeaningTool(),
            TableMeaningTool(),
            ERAnalysisTool(),
            
            # 生成工具
            ScenarioTool(),
            OperationSelectionTool(),
            QuestionGenerationTool(llm=self.llm),
            SQLGenerationTool(llm=self.llm),
            
            # 验证工具
            SQLValidationTool(db_manager=db_manager),
            SQLExecutionTool(db_manager=db_manager),
            
            # 反思工具
            SQLReflectionTool(llm=self.llm),
            SequentialThinkingTool(llm=self.llm)
        ]
        
        return tools
    
    def _initialize_memory(self) -> BaseMemory:
        """初始化记忆系统"""
        return DatabaseAnalysisMemory()
    
    def generate_training_data(self, count: int, output_file: str) -> Dict[str, Any]:
        """生成训练数据"""
        # 使用 Agent 执行生成任务
        result = self.agent_executor.run(
            f"生成{count}个SQL训练数据，保存到{output_file}"
        )
        return result
```

from jinja2 import Environment, FileSystemLoader
from langchain.prompts import ChatPromptTemplate, SystemMessagePromptTemplate
from typing import Dict, Any

class PromptManager:
    """提示词管理器"""
    
    def __init__(self):
        self.env = Environment(
            loader=FileSystemLoader('prompts/templates')
        )
    
    def get_system_prompt(self) -> str:
        """获取系统提示词"""
        template = self.env.get_template('system/main.j2')
        return template.render()
    
    def get_tool_prompt(self, tool_name: str) -> str:
        """获取工具提示词"""
        template = self.env.get_template(f'tools/{tool_name}.j2')
        return template.render()
    
    def create_agent_prompt(self) -> ChatPromptTemplate:
        """创建Agent提示词模板"""
        system_prompt = self.get_system_prompt()
        
        return ChatPromptTemplate.from_messages([
            SystemMessagePromptTemplate.from_template(system_prompt),
            ("human", "{input}"),
            ("assistant", "{agent_scratchpad}")
        ])
```
