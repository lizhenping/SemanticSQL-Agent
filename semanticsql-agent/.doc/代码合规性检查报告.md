## 建议的改进

### 1. 完善数据提取逻辑

在 `sql_agent.py` 的 `_extract_generated_examples` 方法中：
```python
def _extract_generated_examples(self, agent_output: Any) -> List[Dict[str, Any]]:
    """从 Agent 输出中提取生成的样例"""
    examples = []
    
    # 从执行轨迹中提取数据
    if hasattr(self.callback_handler, 'get_trajectories'):
        trajectories = self.callback_handler.get_trajectories()
        
        # 解析轨迹，提取问题-SQL对
        current_example = {}
        for trajectory in trajectories:
            if trajectory['type'] == 'action':
                tool = trajectory['tool']
                if tool == 'question_generation':
                    current_example['question'] = trajectory['output']['question']
                elif tool == 'sql_generation':
                    current_example['sql'] = trajectory['output']['sql']
                elif tool == 'sql_execution':
                    current_example['validation'] = trajectory['output']
                    # 完成一个样例
                    if 'question' in current_example and 'sql' in current_example:
                        examples.append(current_example)
                        current_example = {}
    
    return examples
```
