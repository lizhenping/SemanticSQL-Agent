try:
    result = domain_tool._run()

### 2. 异步分析

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor

async def analyze_database_async(db_name):
    """异步分析单个数据库"""
    loop = asyncio.get_event_loop()
    
    def _analyze():
        db_manager.switch_database(db_name) 
        tool = create_domain_analysis_tool(
            memory_manager=neo4j_manager,
            database_manager=db_manager
        )
        return tool._run()
    
    # 在线程池中执行
    with ThreadPoolExecutor() as executor:
        result = await loop.run_in_executor(executor, _analyze)
    
    return db_name, result

# 并发分析多个数据库
async def batch_analyze():
    databases = ["db1", "db2", "db3", "db4"]
    tasks = [analyze_database_async(db) for db in databases]
    results = await asyncio.gather(*tasks)
    return dict(results)

# 执行
results = asyncio.run(batch_analyze())
```

### 1. 性能监控

```python
import time
from contextlib import contextmanager

@contextmanager
def performance_monitor(operation_name):
    """性能监控上下文管理器"""
    start_time = time.time()
    try:
        yield
    finally:
        duration = time.time() - start_time
        print(f"{operation_name} 耗时: {duration:.2f}秒")

# 使用监控
with performance_monitor("领域分析"):
    result = domain_tool._run()

# 详细的性能分析
def analyze_performance():
    """分析性能瓶颈"""
    metrics = {}
    
    # Neo4j查询性能
    start = time.time()
    schema_data = domain_tool._query_neo4j_schema()
    metrics['neo4j_query'] = time.time() - start
    
    # DDL格式化性能
    start = time.time()
    ddl_content = domain_tool._format_schema_to_ddl(schema_data)
    metrics['ddl_formatting'] = time.time() - start
    
    # LLM分析性能
    start = time.time()
    domain_knowledge = domain_tool._analyze_domain_with_llm(ddl_content)
    metrics['llm_analysis'] = time.time() - start
    
    return metrics

performance_data = analyze_performance()
for operation, duration in performance_data.items():
    print(f"{operation}: {duration:.2f}秒")
```

## 最佳实践总结

### 1. 使用建议

1. **依赖管理**: 确保schema_extraction_tool先执行
2. **缓存策略**: 对稳定的数据库启用缓存
3. **错误处理**: 实现完整的异常处理逻辑
4. **性能监控**: 定期监控分析性能
5. **结果验证**: 建立结果质量检查机制

### 2. 常见问题解决

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| 分析置信度低 | Schema信息不完整 | 检查数据库连接和权限 |
| LLM调用超时 | 网络或服务问题 | 增加超时时间或使用降级分析 |
| Neo4j写入失败 | 权限或连接问题 | 检查Neo4j配置和网络连接 |
| 依赖检查失败 | schema_extraction_tool未执行 | 先执行前置工具 |

通过这些详细的使用示例，用户可以快速上手Domain Analysis Tool，并在各种实际场景中发挥其强大的业务分析能力。
