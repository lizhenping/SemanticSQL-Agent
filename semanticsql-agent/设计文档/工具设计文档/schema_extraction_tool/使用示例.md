from typing import List, Dict, Any

def process_multiple_databases(database_configs: List[Dict[str, Any]], 
                             neo4j_config: Dict[str, Any]):
    """处理多个数据库的Schema提取"""
    
    # 共享Neo4j连接
    memory_manager = Neo4jMemoryManager(neo4j_config)
    
    results = []
    for db_config in database_configs:
        try:
            # 为每个数据库创建连接
            db_manager = DatabaseManager(db_config)
            
            # 创建工具
            tool = create_schema_extraction_tool(
                memory_manager=memory_manager,
                database_manager=db_manager
            )
            
            # 执行提取
            result = tool._run()
            results.append({
                "database": db_config["database"],
                "status": "success",
                "message": result
            })
            
        except Exception as e:
            results.append({
                "database": db_config["database"],
                "status": "error", 
                "message": str(e)
            })
        finally:
            # 清理连接
            if 'db_manager' in locals():
                db_manager.close()
    
    return results

# 使用示例

### 2. 增量更新场景

```python
class IncrementalSchemaExtractionTool(SchemaExtractionTool):
    def __init__(self, last_update_time=None, **kwargs):
        super().__init__(**kwargs)
        self.last_update_time = last_update_time
    
    def _should_process_table(self, table_name: str) -> bool:
        """检查表是否需要更新"""
        if not self.last_update_time:
            return True
        
        # 查询Neo4j中表的最后更新时间
        if self.memory_manager and self.memory_manager.neo4j_graph:
            cypher = """
            MATCH (t:Table {name: $table_name})
            RETURN t.last_updated as last_updated
            """
            result = self.memory_manager.neo4j_graph.query(cypher, {"table_name": table_name})
            
            if result and len(result) > 0:
                last_updated = result[0].get("last_updated")
                if last_updated and last_updated >= self.last_update_time:
                    return False  # 无需更新
        
        return True
    
    def _extract_mysql_metadata(self, db_manager):
        """增量提取元数据"""
        # 获取基础数据
        raw_data = super()._extract_mysql_metadata(db_manager)
        
        # 过滤需要更新的表
        tables_to_process = []
        for table_data in raw_data["tables_data"]:
            if self._should_process_table(table_data["name"]):
                tables_to_process.append(table_data)
            else:
                self.logger.info(f"跳过表 {table_data['name']}，无需更新")
        
        raw_data["tables_data"] = tables_to_process
        return raw_data

# 增量更新使用
from datetime import datetime, timedelta

last_update = datetime.now() - timedelta(days=1)
incremental_tool = IncrementalSchemaExtractionTool(
    last_update_time=last_update,
    memory_manager=memory_manager,
    database_manager=db_manager
)
```

### 3. 数据质量监控

class QualityAwareSchemaExtractionTool(SchemaExtractionTool):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.quality_metrics = {
            "total_tables": 0,
            "tables_with_comments": 0,
            "total_columns": 0,
            "columns_with_comments": 0,
            "foreign_keys_detected": 0,
            "high_entropy_columns": 0
        }
    
    def _extract_table_metadata(self, db_manager, table_name):
        """提取表元数据并收集质量指标"""
        table_data = super()._extract_table_metadata(db_manager, table_name)
        
        # 更新质量指标
        self.quality_metrics["total_tables"] += 1
        if table_data.get("business_desc"):
            self.quality_metrics["tables_with_comments"] += 1
        
        for column in table_data.get("columns", []):
            self.quality_metrics["total_columns"] += 1
            if column.get("business_desc"):
                self.quality_metrics["columns_with_comments"] += 1
            if column.get("is_foreign"):
                self.quality_metrics["foreign_keys_detected"] += 1
            if column.get("entropy_level") == "high":
                self.quality_metrics["high_entropy_columns"] += 1
        
        return table_data
    
    def _run(self, *args, **kwargs):
        """执行提取并生成质量报告"""
        result = super()._run(*args, **kwargs)
        
        # 生成质量报告
        self._generate_quality_report()
        
        return result
    
    def _generate_quality_report(self):
        """生成数据质量报告"""
        metrics = self.quality_metrics
        
        comment_coverage_tables = (metrics["tables_with_comments"] / metrics["total_tables"] * 100) if metrics["total_tables"] > 0 else 0
        comment_coverage_columns = (metrics["columns_with_comments"] / metrics["total_columns"] * 100) if metrics["total_columns"] > 0 else 0
        
        quality_report = f"""

📋 基础统计:
   • 总表数: {metrics['total_tables']}
   • 总列数: {metrics['total_columns']}
   
💬 注释覆盖率:
   • 表注释覆盖: {comment_coverage_tables:.1f}% ({metrics['tables_with_comments']}/{metrics['total_tables']})
   • 列注释覆盖: {comment_coverage_columns:.1f}% ({metrics['columns_with_comments']}/{metrics['total_columns']})
   
🔗 关系检测:
   • 外键检测: {metrics['foreign_keys_detected']} 个
   
📈 数据分析:
   • 高熵列: {metrics['high_entropy_columns']} 个
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
        """
        
        self.logger.info(quality_report)

    memory_manager=memory_manager,

)

class RobustSchemaExtractionTool(SchemaExtractionTool):
    def __init__(self, max_retries=3, **kwargs):
        super().__init__(**kwargs)
        self.max_retries = max_retries
    
    def _extract_table_metadata_with_retry(self, db_manager, table_name):
        """带重试的表提取"""
        last_error = None
        
        for attempt in range(self.max_retries):
            try:
                return self._extract_table_metadata(db_manager, table_name)
            except Exception as e:
                last_error = e
                self.logger.warning(f"表 {table_name} 提取失败 (尝试 {attempt + 1}/{self.max_retries}): {e}")
                
                if attempt < self.max_retries - 1:
                    time.sleep(1 * (attempt + 1))  # 递增延迟
        
        # 所有重试都失败，返回基础信息
        self.logger.error(f"表 {table_name} 提取最终失败: {last_error}")
        return {
            "name": table_name,
            "row_count": None,
            "business_desc": "",
            "columns": []
        }
    
    def _calculate_entropy_level_safe(self, db_manager, table_name, column_name):
        """安全的熵值计算"""

### 2. 并行处理

```python
import concurrent.futures
from typing import List

class ParallelSchemaExtractionTool(SchemaExtractionTool):
    def __init__(self, max_workers=4, **kwargs):
        super().__init__(**kwargs)
        self.max_workers = max_workers
    
    def _extract_tables_parallel(self, db_manager, table_names: List[str]):
        """并行提取多个表的元数据"""
        tables_data = []
        
        with concurrent.futures.ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            # 提交所有任务
            future_to_table = {
                executor.submit(self._extract_table_metadata, db_manager, table_name): table_name
                for table_name in table_names
            }
            
            # 收集结果
            for future in concurrent.futures.as_completed(future_to_table):
                table_name = future_to_table[future]
                try:
                    table_data = future.result(timeout=60)  # 60秒超时
                    tables_data.append(table_data)
                    self.logger.debug(f"表 {table_name} 处理完成")
                except Exception as e:
                    self.logger.error(f"表 {table_name} 处理失败: {e}")
        
        return tables_data

# 使用并行工具（注意：需要确保数据库连接线程安全）
parallel_tool = ParallelSchemaExtractionTool(
    max_workers=4,
    memory_manager=memory_manager,
    database_manager=db_manager
)
```
