# 工具基类接口设计 - 极简版

基于最新的 06-ReAct智能体接口设计，工具应该直接返回结果给 ReAct Observation，采用极简设计原则。

## 1. 记忆数据模型 (tools/schemas.py)

### 1.1 语义三元组数据模型
```python
from pydantic import BaseModel, Field
from typing import Optional, Tuple
from langchain_neo4j import Neo4jGraph
from langchain_community.graphs.graph_document import GraphDocument, Node, Relationship
from enum import Enum

class PredicateType(Enum):
    """谓词类型枚举 - 统一使用英文"""
    HAS_COLUMN = "has_column"
    HAS_TABLE = "has_table"
    HAS_RELATIONSHIP = "has_relationship"
    HAS_CONSTRAINT = "has_constraint"
    HAS_INDEX = "has_index"
    BELONGS_TO = "belongs_to"
    REFERENCES = "references"
    CONTAINS = "contains"
    DERIVED_FROM = "derived_from"
    SIMILAR_TO = "similar_to"

class SemanticTriple(BaseModel):
    """语义三元组模型 - 工具记忆的核心数据结构"""
    subject: str = Field(description="主体实体")
    predicate: str = Field(description="关系谓词")
    object: str = Field(description="客体实体")
    
    # 扩展属性
    subject_type: str = Field(default="Entity", description="主体类型")
    object_type: str = Field(default="Entity", description="客体类型") 
    confidence: Optional[float] = Field(default=None, description="置信度（可选）")
    source_tool: str = Field(default="", description="来源工具")

class MemoryData(BaseModel):
    """记忆数据模型 - 标准化的知识表示"""
    
    subject: str = Field(description="主语")
    predicate: str = Field(description="谓语/关系")
    object: str = Field(description="宾语")
    
    # 元数据字段
    confidence: float = Field(default=1.0, description="置信度")
    source: str = Field(description="数据来源")
    timestamp: str = Field(description="时间戳")
    session_id: str = Field(description="会话ID")
    
    def to_simple_tuple(self) -> Tuple[str, str, str]:
        """转换为简单三元组，保持向后兼容"""
        return (self.subject, self.predicate, self.object)
```

## 2. 极简工具基类 (tools/base_tool.py)

### 2.1 极简化的工具基类 - 只保留核心功能
```python
from langchain.tools import BaseTool
from typing import Optional, List
from .schemas import SemanticTriple

class BaseSemanticSQLTool(BaseTool):
    """SemanticSQL工具基类 - 极简设计，只提供核心方法"""
    
    def __init__(self, neo4j_graph: Optional[Neo4jGraph] = None, **kwargs):
        super().__init__(**kwargs)
        self.neo4j_graph = neo4j_graph  # 可选的Neo4j图实例
        self._generated_triples = []     # 存储当前执行生成的三元组
    
    def _run(self, input_text: str) -> str:
        """
        工具执行入口 - 子类必须重写此方法实现所有业务逻辑
        
        Args:
            input_text: ReAct Agent传入的Action Input文本
            
        Returns:
            工具自定义的返回内容，直接传给ReAct Observation
        """
        self._generated_triples = []
        raise NotImplementedError("Subclasses must implement _run method")
    
    def get_memory_by_source_tool(self, source_tool: str, limit: int = 10) -> List[dict]:
        """
        获取指定工具生成的记忆三元组 - 唯一的记忆查询方法
        
        Args:
            source_tool: 来源工具名称
            limit: 返回数量限制
            
        Returns:
            指定工具生成的三元组列表
        """
        if not self.neo4j_graph:
            return []
        
        try:
            cypher_query = f"""
            MATCH (s)-[r]->(o)
            WHERE r.source_tool = '{source_tool}'
            RETURN s.name as subject, type(r) as predicate, o.name as object,
                   r.confidence as confidence, r.source_tool as source_tool
            ORDER BY id(r) DESC
            LIMIT {limit}
            """
            
            results = self.neo4j_graph.query(cypher_query)
            return results
            
        except Exception as e:
            print(f"查询工具记忆失败: {e}")
            return []
    
    def add_analysis_triple(self, 
                           subject: str,
                           predicate: str, 
                           object: str,
                           subject_type: str = "Entity",
                           object_type: str = "Entity",
                           confidence: Optional[float] = None) -> None:
        """
        添加分析三元组到当前工具记忆 - 唯一的三元组添加方法
        
        Args:
            subject: 主体实体
            predicate: 关系谓词  
            object: 客体实体
            subject_type: 主体类型
            object_type: 客体类型
            confidence: 置信度
        """
        triple = SemanticTriple(
            subject=subject,
            predicate=predicate,
            object=object,
            subject_type=subject_type,
            object_type=object_type,
            confidence=confidence,
            source_tool=self.name
        )
        self._generated_triples.append(triple)
    
    # Neo4j集成方法（内部使用）
    def convert_to_graph_document(self, triples: List[SemanticTriple], 
                                 source: str = "SemanticSQL-Agent") -> GraphDocument:
        """将语义三元组转换为LangChain GraphDocument"""
        nodes = []
        relationships = []
        node_set = set()
        
        for triple in triples:
            # 创建主体节点
            if triple.subject not in node_set:
                subject_node = Node(
                    id=triple.subject,
                    type=triple.subject_type,
                    properties={"name": triple.subject, "source": triple.source_tool}
                )
                nodes.append(subject_node)
                node_set.add(triple.subject)
            
            # 创建客体节点
            if triple.object not in node_set:
                object_node = Node(
                    id=triple.object,
                    type=triple.object_type,
                    properties={"name": triple.object, "source": triple.source_tool}
                )
                nodes.append(object_node)
                node_set.add(triple.object)
            
            # 创建关系
            relationship = Relationship(
                source=Node(id=triple.subject, type=triple.subject_type),
                target=Node(id=triple.object, type=triple.object_type),
                type=triple.predicate.upper().replace(" ", "_"),
                properties={
                    "confidence": triple.confidence,
                    "source_tool": triple.source_tool
                }
            )
            relationships.append(relationship)
        
        return GraphDocument(
            nodes=nodes,
            relationships=relationships,
            source=source
        )
    
    def persist_triples_to_neo4j(self, triples: List[SemanticTriple]) -> None:
        """将语义三元组持久化到Neo4j - 内部方法，供工具调用"""
        if not triples or not self.neo4j_graph:
            return
        
        graph_doc = self.convert_to_graph_document(triples)
        self.neo4j_graph.add_graph_documents([graph_doc])
```

## 3. 使用示例

### 3.1 基础工具实现 - 完全自定义
```python
from tools.base_tool import BaseSemanticSQLTool

class SchemaExtractionTool(BaseSemanticSQLTool):
    """数据库结构提取工具 - 极简实现"""
    
    name: str = "schema_extraction"
    description: str = "分析数据库结构，提取表和字段信息"
    
    def _run(self, input_text: str) -> str:
        """在_run中实现所有逻辑"""
        # 1. 清空上次执行的三元组
        self._generated_triples = []
        
        # 2. 业务逻辑：解析输入参数（模拟数据库连接）
        mock_tables = {
            "users": ["id", "username", "email", "created_at"],
            "orders": ["id", "user_id", "product_id", "amount", "order_date"],
            "products": ["id", "name", "price", "category_id"]
        }
        
        # 3. 生成数据库结构三元组
        for table_name, columns in mock_tables.items():
            # 添加表与数据库的关系
            self.add_analysis_triple(
                subject="电商数据库",
                predicate=PredicateType.HAS_TABLE.value, 
                object=table_name,
                subject_type="Database",
                object_type="Table",
                confidence=1.0
            )
            
            # 添加表与字段的关系
            for column in columns:
                self.add_analysis_triple(
                    subject=table_name,
                    predicate=PredicateType.HAS_COLUMN.value,
                    object=column,
                    subject_type="Table",
                    object_type="Column",
                    confidence=0.95
                )
        
        # 4. 存储到Neo4j（如果需要）
        if self._generated_triples and self.neo4j_graph:
            self.persist_triples_to_neo4j(self._generated_triples)
        
        # 示例：存储表信息到记忆
        from datetime import datetime
        table_memory = MemoryData(
            subject="users_table",
            predicate=PredicateType.HAS_COLUMN.value,
            object="id,username,email,created_at",
            source=self.name,
            timestamp=datetime.now().isoformat(),
            session_id="current_session"
        )
        
        # 5. 完全自定义返回内容
        return f"✅ 数据库结构分析完成：发现 {len(mock_tables)} 个表，共 {sum(len(cols) for cols in mock_tables.values())} 个字段"

# 使用方式：
schema_tool = SchemaExtractionTool()
result = schema_tool._run("分析电商数据库")
print(result)  # 输出：✅ 数据库结构分析完成：发现 3 个表，共 13 个字段
```

## 4. 极简设计的优势

### 4.1 核心优势
1. **真正极简**：基类只有2个核心方法，无过度设计
2. **完全自主**：工具自定义所有逻辑、存储时机、返回格式
3. **记忆分片**：每个工具管理自己的记忆片段，依赖关系清晰
4. **LLM友好**：工具内部直接处理原始数据，无需预格式化

### 4.2 工具开发优势
1. **开发简单**：只需实现一个`_run`方法
2. **逻辑清晰**：所有逻辑都在`_run`中，易于理解和调试
3. **返回自由**：完全控制返回给ReAct Observation的内容
4. **存储灵活**：可选择性存储到Neo4j，满足不同需求

### 4.3 系统优势
1. **性能高效**：去除不必要的方法调用和格式转换
2. **维护简单**：基类代码量极少，稳定可靠
3. **扩展容易**：新工具只需继承基类，实现`_run`方法
4. **ReAct兼容**：完全符合最新ReAct智能体设计理念

---

**总结**：这套极简设计完全符合"工具执行→存储记忆→自定义返回"的理想流程，去除了所有不必要的复杂度，让工具开发回归本质：专注业务逻辑，自主控制输出。
